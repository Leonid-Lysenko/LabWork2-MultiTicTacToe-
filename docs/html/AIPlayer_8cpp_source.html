52 Position3D AIPlayer::get3DMove(const Board3D& board) {

55  if (winMove.level != -1) {

82 Position3D AIPlayer::findWinningMove(const Board3D& board, Cell player) const {
83  for (int level = 0; level < Board3D::LEVELS; ++level) {
84  for (int row = 0; row < Board::SIZE; ++row) {
85  for (int col = 0; col < Board::SIZE; ++col) {
86  if (board.getLevel(level).getCell(row, col) == Cell::EMPTY) {
87  // Create a temporary board to simulate the move
88  Board3D tempBoard = board;
89  tempBoard.makeMove(level, Move(Position(row, col), player));
90  // Check if this move results in a win
91  if (tempBoard.checkWin(player)) {
92  return Position3D(level, Position(row, col));
93  }
94  }
95  }
96  }
97  }

99  return Position3D(-1, Position(-1, -1));
